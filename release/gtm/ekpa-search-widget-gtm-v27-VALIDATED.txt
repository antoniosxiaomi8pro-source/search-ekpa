  "use strict";
  var BACKEND_URL = "https://nodejs-production-5371.up.railway.app"; // hardcoded - see header note on GTM Variable bug
  var GA4_ID      = "G-ZL8304SEHG"; // Elearning - GA4
  var SELECTORS   = ["#qHeroFront", "#qNav", "#qNavMobile", "#qAside", "#qSearch"];  // confirmed against the real DOM (v14: header/mobile + category-aside; v15: desktop top-nav; v19: /search results page)

  var MIN_CHARS   = 2;
  var DEBOUNCE_MS = 220;
  var MAX_RESULTS = 6;
  var RELATED_MAX = 8; // v26: related programs shown per result card (was 3)
  BACKEND_URL = String(BACKEND_URL || "").replace(/\/+$/, "");
  var EARLY_ID = "ekpa-sf-early-hide";
  function earlyShow() {
    try {
      var st = document.getElementById(EARLY_ID);
      if (st && st.parentNode) st.parentNode.removeChild(st);
    } catch (e) {}
  }
  // /search?q=... : hide the website's own result list right away (before the
  // backend answers) so it never flashes. Always undone by earlyShow().
  function earlyHide() {
    try {
      if (window.location.pathname !== "/search") return;
      var m = /(?:^|[?&])q=([^&]*)/.exec(window.location.search || "");
      if (!m) return;
      var q = decodeURIComponent(m[1].replace(/\+/g, " ")).trim();
      if (ltFn(q.length, MIN_CHARS)) return;
      if (document.getElementById(EARLY_ID)) return;
      var st = document.createElement("style");
      st.id = EARLY_ID;
      st.textContent =
        "main.content--search .container--search-results{display:none !important;}" +
        ".search-overview__title,.search-overview__query,.search-overview__tagline{visibility:hidden !important;}" +
        "main.content--search .section--search-header:after{content:\"\u03a6\u03cc\u03c1\u03c4\u03c9\u03c3\u03b7 \u03b1\u03c0\u03bf\u03c4\u03b5\u03bb\u03b5\u03c3\u03bc\u03ac\u03c4\u03c9\u03bd\u2026\";display:block;text-align:center;color:#667085;padding:32px 0;}";
      (document.head || document.documentElement).appendChild(st);
      setTimeout(earlyShow, 6000);
    } catch (e) {}
  }
  earlyHide();
  function ga4(eventName, params) {
    try {
      if (typeof window.gtag === "function") {
        window.gtag("event", eventName, Object.assign({ send_to: GA4_ID }, params));
      } else {
        window.dataLayer = window.dataLayer || [];
        window.dataLayer.push(Object.assign({ event: "ekpa_" + eventName }, params));
      }
    } catch (e) {}
  }
  function esc(v) {
    return String(v == null ? "" : v)
      .replace(/&/g, "&amp;").replace(/\x3C/g, "&lt;")
      .replace(/\x3E/g, "&gt;").replace(/"/g, "&quot;");
  }
  function ltFn(a, b) { return Math.sign(a - b) === -1; }
  function gtFn(a, b) { return Math.sign(a - b) === 1; }
  function injectHoverStyle() {
    if (document.getElementById("ekpa-tt-hover-style")) return; // once per page
    try {
      var styleTag = document.createElement("style");
      styleTag.id = "ekpa-tt-hover-style";
      styleTag.type = "text/css";
      var css = ".tt-suggestion:hover, .tt-suggestion.tt-cursor { background-color: #f5f7fa !important; cursor: pointer !important; }";
      if (styleTag.styleSheet) { styleTag.styleSheet.cssText = css; } else { styleTag.appendChild(document.createTextNode(css)); }
      (document.head || document.documentElement).appendChild(styleTag);
    } catch (e) {}
  }
  function debounce(fn, ms) {
    var t;
    return function () {
      var args = arguments, ctx = this;
      clearTimeout(t);
      t = setTimeout(function () { fn.apply(ctx, args); }, ms);
    };
  }
  function programKey(p) {
    return p ? (p.slug || (p.id != null ? String(p.id) : "")) : "";
  }
  var BACKEND_READY = /^https?:\/\//.test(BACKEND_URL);
  if (!BACKEND_READY) {
    try { console.warn("[EKPA widget] BACKEND_URL is not configured correctly. Widget is inactive."); } catch (e) {}
  }
  var requestSeq = 0;
  var currentController = null;
  function fetchResults(query, cb) {
    if (!BACKEND_READY) { cb([], true); return; }
    var mySeq = ++requestSeq;
    if (currentController) { try { currentController.abort(); } catch (e) {} }
    currentController = typeof AbortController === "function" ? new AbortController() : null;
    fetch(BACKEND_URL + "/api/search?q=" + encodeURIComponent(query) + "&limit=" + MAX_RESULTS + "&fields=compact", currentController ? { signal: currentController.signal } : undefined)
      .then(function (r) { return r.ok ? r.json() : null; })
      .then(function (data) {
        if (mySeq !== requestSeq) return; // stale response
        cb(Array.isArray(data) ? data : [], !Array.isArray(data));
      })
      .catch(function () {
        if (mySeq !== requestSeq) return;
        cb([], true);
      });
  }
  function trackClick(key, q) {
    if (!BACKEND_READY || !key) return;
    var url = BACKEND_URL + "/api/track-click";
    var payload = JSON.stringify({ query: q, program_id: key });
    try {
      if (navigator.sendBeacon && navigator.sendBeacon(url, new Blob([payload], { type: "text/plain;charset=UTF-8" }))) return;
      fetch(url, { method: "POST", headers: { "Content-Type": "text/plain;charset=UTF-8" }, body: payload, keepalive: true })["catch"](function () {});
    } catch (e) {}
  }
  var lastNoResultsQuery = "";
  function reportNoResults(query) {
    if (query === lastNoResultsQuery) return; // once per query
    lastNoResultsQuery = query;
    ga4("search_no_results", { search_term: query });
  }
  function reportSelect(p, query) {
    ga4("select_content", {
      content_type: "course",
      item_id: p.slug || p.url,
      content_id: p.title,
      search_term: query,
      transport_type: "beacon"
    });
    trackClick(programKey(p), query);
  }
  function tryTypeahead(input) {
    var $ = window.jQuery;
    if (!$ || !$.fn || !$.fn.typeahead) return false;
    var $input = $(input);
    try { $input.typeahead("destroy"); } catch (e) {}
    try {
      var pendingTimer = null;
      $input.typeahead(
        { hint: false, highlight: true, minLength: MIN_CHARS },
        {
          name: "ekpa-programs",
          limit: 100,
          source: function (query, syncResults, asyncResults) {
            clearTimeout(pendingTimer);
            pendingTimer = setTimeout(function () {
              fetchResults(query, function (data, failed) {
                if (!data.length && !failed) reportNoResults(query);
                asyncResults(data.slice(0, MAX_RESULTS));
              });
            }, DEBOUNCE_MS);
          },
          display: function (item) { return item.title; },
          templates: {
            suggestion: function (item) {
              return (
                '\x3Cdiv style="' + SITE_FONT + 'display:flex;gap:10px;align-items:flex-start;padding:8px 0;text-align:left"\x3E' +
                '\x3Cimg src="' + esc(item.image_url || "") + '" alt="" style="width:100px;height:70px;object-fit:cover;border-radius:6px;background:#eef2f6;flex-shrink:0" onerror="this.style.visibility=\'hidden\'"\x3E' +
                '\x3Cdiv style="flex:1;min-width:0"\x3E' +
                '\x3Cdiv style="color:#17212b;white-space:normal;word-break:break-word;line-height:1.35"\x3E' + esc(item.title) + "\x3C/div\x3E" +

                "\x3C/div\x3E\x3C/div\x3E"
              );
            },
            notFound: '\x3Cdiv style="' + SITE_FONT + 'padding:10px 14px;color:#667085;font-size:13px"\x3E\u0394\u03b5\u03bd \u03b2\u03c1\u03ad\u03b8\u03b7\u03ba\u03b1\u03bd \u03b1\u03c0\u03bf\u03c4\u03b5\u03bb\u03ad\u03c3\u03bc\u03b1\u03c4\u03b1\x3C/div\x3E'
          }
        }
      );
      var typedQuery = "";
      $input.off("input.ekpa").on("input.ekpa", function () { typedQuery = input.value; });
      $input.off("typeahead:select.ekpa").on("typeahead:select.ekpa", function (ev, item) {
        if (!item) return;
        reportSelect(item, typedQuery || input.value);
        if (item.url) window.location.href = item.url;
      });
      var menuEl = $input.parent().find(".tt-menu")[0];
      var styleMenu = function () {
        if (!menuEl) return;
        menuEl.style.cssText = menuEl.style.cssText +
          ";background:#fff;border:1px solid #dfe5ec;border-radius:10px;" +
          "box-shadow:0 12px 30px rgba(16,24,40,.15);z-index:9999;" +
          "max-height:360px;overflow:auto";
        // The host page ships its own CSS for .tt-dataset/.tt-suggestion (from
        // before this widget existed) that centers text and adds large
        // padding/margins. Override it inline on every render so our own
        // template's layout (left-aligned, compact spacing) always wins,
        // regardless of what the page's stylesheet does.
        var datasetEls = menuEl.querySelectorAll(".tt-dataset");
        for (var di = 0; ltFn(di, datasetEls.length); di++) {
          datasetEls[di].style.cssText = datasetEls[di].style.cssText + ";padding:0;margin:0;";
        }
        var suggestionEls = menuEl.querySelectorAll(".tt-suggestion");
        for (var si = 0; ltFn(si, suggestionEls.length); si++) {
          suggestionEls[si].style.cssText = suggestionEls[si].style.cssText +
            ";text-align:left;padding:0;margin:0;";
        }
      };
      if (menuEl) {
        styleMenu();
        $input.on("typeahead:render.ekpa", styleMenu);
        $input.on("typeahead:open.ekpa", styleMenu);
      }
      return true;
    } catch (e) { return false; }
  }
  function buildDropdown(input) {
    var wrap = document.createElement("div");
    wrap.style.position = "relative";
    input.parentNode.insertBefore(wrap, input);
    wrap.appendChild(input);
    var dd = document.createElement("div");
    dd.style.cssText =
      "position:absolute;top:100%;left:0;right:0;background:#fff;" +
      "border:1px solid #dfe5ec;border-radius:10px;" +
      "box-shadow:0 12px 30px rgba(16,24,40,.15);margin-top:6px;" +
      "z-index:9999;max-height:360px;overflow:auto;display:none;text-align:left";
    wrap.appendChild(dd);
    return dd;
  }
  function renderDropdown(dd, results, query, failed) {
    if (!results.length) {
      dd.style.display = "none";
      if (!failed) reportNoResults(query); // network error != "0 results"
      return;
    }
    var shown = results.slice(0, MAX_RESULTS);
    dd.innerHTML = shown.map(function (p, i) {

  // Symfony's own /search page appears to require every typed word to match
  // (AND logic) - confirmed live: "\u03c0\u03c1\u03bf\u03b3\u03c1\u03ac\u03bc\u03bc\u03b1\u03c4\u03b1 \u03bc\u03ac\u03c1\u03ba\u03b5\u03c4\u03b9\u03bd\u03b3\u03ba" (a generic word plus the real
  // term) returns zero results there, while the bare term alone returns 7.
  // SmartFinder itself does not have this problem (concept expansion), but
  // Enter/icon-click still submits to Symfony's own page (see changelog,
  // routing to SmartFinder is a separate, larger decision). As a narrow,
  // low-risk improvement in the meantime: strip a short list of generic
  // filler words from the query right before the form submits, so Symfony's
  // AND-matching has a better chance of finding the real term. Whole-word
  // match only (never a substring inside another word), case-insensitive
  // and accent-insensitive (see stripAccents below); if stripping would
  // empty the query, the original is submitted instead.
  //
  // v21 fix: v20 relied only on a "submit" event listener attached to the
  // form. Live testing showed the query still reached Symfony unstripped.
  // Root cause: the site's own JS intercepts Enter / the search-icon click
  // and navigates on its own (most likely calling something like
  // window.location or form.submit() directly), which either never fires a
  // native "submit" event at all, or reads the input's value before our
  // listener gets a turn. A submit-event listener cannot reliably win that
  // race. Fix: also clean the input's value from document-level, CAPTURE
  // PHASE "keydown" (Enter) and "click" listeners. Capture-phase listeners
  // on document always run before any listener attached directly to the
  // input or its form/button - regardless of script load order - because
  // capture travels top-down from document to the target before the event
  // ever reaches it. That guarantees our cleanup runs first, no matter what
  // the site's own code does afterward. The original submit-listener is
  // kept as a harmless fallback for the plain case (native form submit).
  var FILLER_WORDS = [
    "\u03c0\u03c1\u03cc\u03b3\u03c1\u03b1\u03bc\u03bc\u03b1",
    "\u03c0\u03c1\u03bf\u03b3\u03c1\u03ac\u03bc\u03bc\u03b1\u03c4\u03b1",
    "\u03c0\u03c1\u03bf\u03b3\u03c1\u03ac\u03bc\u03bc\u03b1\u03c4\u03bf\u03c2",
    "\u03c0\u03c1\u03bf\u03b3\u03c1\u03b1\u03bc\u03bc\u03ac\u03c4\u03c9\u03bd"
  ];
  function stripAccents(s) {
    try {
      return String(s || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    } catch (e) {
      return String(s || "");
    }
  }
  function normalizeWord(w) {
    return stripAccents(String(w || "").toLowerCase());
  }
  function stripFillerWords(q) {
    var words = String(q || "").split(/\s+/).filter(function (w) { return gtFn(w.length, 0); });
    var kept = [];
    for (var i = 0; ltFn(i, words.length); i++) {
      var w = words[i], isFiller = false, normW = normalizeWord(w);
      for (var j = 0; ltFn(j, FILLER_WORDS.length); j++) {
        if (normW === normalizeWord(FILLER_WORDS[j])) { isFiller = true; break; }
      }
      if (!isFiller) kept.push(w);
    }
    var cleaned = kept.join(" ");
    return cleaned.length ? cleaned : q;
  }
  function cleanInputValue(input) {
    try {
      var cleaned = stripFillerWords(input.value);
      if (cleaned !== input.value) input.value = cleaned;
    } catch (e) {}
  }

  function greekSlug(s) {
    var map = {
      "\u03b1":"a","\u03b2":"b","\u03b3":"g","\u03b4":"d","\u03b5":"e","\u03b6":"z","\u03b7":"i","\u03b8":"th",
      "\u03b9":"i","\u03ba":"k","\u03bb":"l","\u03bc":"m","\u03bd":"n","\u03be":"x","\u03bf":"o","\u03c0":"p",
      "\u03c1":"r","\u03c3":"s","\u03c2":"s","\u03c4":"t","\u03c5":"u","\u03c6":"f","\u03c7":"ch","\u03c8":"ps","\u03c9":"o"
    };
    var x = String(s || "").toLowerCase();
    try { x = x.normalize("NFD").replace(/[\u0300-\u036f]/g, ""); } catch (e) {}
    x = x.replace(/[\u03b1-\u03c9]/g, function (c) { return map[c] || c; });
    return x.replace(/&/g, " kai ").replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "");
  }

  function loadMoreInPageResults(host, query, page, programIndex, moreWrap, moreButton, sortMode) {
    if (!host || !moreButton || moreButton.disabled) return;

    moreButton.disabled = true;
    moreButton.textContent = "Φόρτωση...";

    var mySeq = inPageRequestSeq;

    if (loadMoreController) {
      try { loadMoreController.abort(); } catch (e) {}
    }

    loadMoreController =
      typeof AbortController === "function" ? new AbortController() : null;

    fetch(
      BACKEND_URL + "/api/search?q=" + encodeURIComponent(query) +
      "&format=paged&page=" + encodeURIComponent(page) +
      "&page_size=40&sort=" + encodeURIComponent(sortMode || "relevance"),
      loadMoreController ? { signal: loadMoreController.signal } : undefined
    )
      .then(function (r) {
        return r.ok ? r.json() : null;
      })
      .then(function (data) {
        if (mySeq !== inPageRequestSeq) return;

        if (!data || !Array.isArray(data.results) || !data.pagination) {
          throw new Error("Invalid paged response");
        }

        if (host.getAttribute("data-smartfinder-query") !== query) return;
        if (host.getAttribute("data-smartfinder-sort") !== (sortMode || "relevance")) return;

        appendResultRows(host, query, data.results, programIndex || {}, moreWrap);

        if (data.pagination.has_next) {
          moreButton.disabled = false;
          moreButton.textContent = "Εμφάνιση περισσότερων";
          moreButton.setAttribute(
            "data-next-page",
            String(data.pagination.page + 1)
          );
        } else if (moreWrap && moreWrap.parentNode) {
          moreWrap.parentNode.removeChild(moreWrap);
        }
      })
      .catch(function () {
        moreButton.disabled = false;
        moreButton.textContent = "Δοκιμάστε ξανά";
      });
  }

        };
        renderInEkpaSearchPage(query, results, parts[1] || {}, pagination, sortMode);
      })
      .catch(function () {
        if (mySeq !== inPageRequestSeq) return;

        setNativeSearchMetaVisible(true);
        earlyShow();

        var old = document.getElementById("ekpa-smartfinder-results");
        if (old && old.parentNode) old.parentNode.removeChild(old);
      });
  }

  function goToEkpaResults(input) {
    try {
      var q = String(input.value || "").trim();
      if (ltFn(q.length, MIN_CHARS)) return false;

      // Stay on the EKPA domain. On non-search pages move to the site's own
      // /search route; v23 reads ?q= there and replaces the native result area.
      if (window.location.pathname !== "/search") {
        window.location.href = "/search?q=" + encodeURIComponent(q);
        return true;
      }

      if (window.history && window.history.replaceState) {
        window.history.replaceState(null, "", "/search?q=" + encodeURIComponent(q));
      }
      loadInPageResults(q);
      return true;
    } catch (e) {
      return false;
    }
  }


  function findSubmitClickTarget(el) {
    try {
      if (!el) return null;
      if (el.closest && el.closest(".tt-menu")) return null;
      var form = el.closest ? el.closest("form") : null;
      if (!form) return null;
      var input = form.querySelector("[data-ekpa-enhanced]");
      if (!input || el === input) return null;
      return input;
    } catch (e) {
      return null;
    }
  }

  var captureCleanupInstalled = false;
  function installCaptureCleanup() {
    if (captureCleanupInstalled) return;
    captureCleanupInstalled = true;

    document.addEventListener("keydown", function (e) {
      if (e.key !== "Enter") return;
      var input = findEnhancedInputNear(e.target);
      if (!input) return;
      var wrap = input.closest ? input.closest(".twitter-typeahead") : null;
      if (wrap && wrap.querySelector(".tt-suggestion.tt-cursor")) return;
      if (goToEkpaResults(input)) {
        e.preventDefault();
        e.stopPropagation();
      }
    }, true);

    document.addEventListener("click", function (e) {
      var input = findSubmitClickTarget(e.target);
      if (!input) return;
      if (goToEkpaResults(input)) {
        e.preventDefault();
        e.stopPropagation();
      }
    }, true);
  }

  // Deep-link support: when /search?q=... opens, render SmartFinder results
  // inside the EKPA page after the normal page DOM is available.
  function runSearchPageQuery() {
    if (window.location.pathname !== "/search") return;
    var m = /(?:^|[?&])q=([^&]*)/.exec(window.location.search || "");
    if (!m) return;
    var q = "";
    try { q = decodeURIComponent(m[1].replace(/\+/g, " ")).trim(); } catch (e) {}
    if (ltFn(q.length, MIN_CHARS)) return;
    var qSearch = document.querySelector("#qSearch");
    if (qSearch) qSearch.value = q;
    loadInPageResults(q);
  }

  function enhance(input) {
    if (!input || input.getAttribute("data-ekpa-enhanced")) return;
    input.setAttribute("data-ekpa-enhanced", "1");
    injectHoverStyle();
    attachSearchAnalytics(input);
    attachQueryCleanup(input);
    installCaptureCleanup();
    if (!tryTypeahead(input)) enhanceStandalone(input);
  }
  function init() {
    SELECTORS.forEach(function (sel) {
      var nodes = document.querySelectorAll(sel);
      for (var i = 0; ltFn(i, nodes.length); i++) enhance(nodes[i]);
    });
  }
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", function () {
      init();
      runSearchPageQuery();
    });
  } else {
    init();
    runSearchPageQuery();
  }
  var attempts = 0;
  var watcher = setInterval(function () {
    attempts++;
    init();
    if (gtFn(attempts, 20)) clearInterval(watcher);
  }, 500);
})();
</script>
